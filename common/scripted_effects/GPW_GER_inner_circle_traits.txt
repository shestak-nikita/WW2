# Upgrade helpers that do not depend on has_completed_focus. NFQ_complete_focus
# runs before the native focus flag is necessarily visible inside a completion reward.
GER_upgrade_todt_organization_to_stage_1 = {
	if = {
		limit = {
			has_idea = GER_organization_todt
		}
		swap_ideas = {
			remove_idea = GER_organization_todt
			add_idea = GER_organization_todt_stage_1
		}
	}
}

GER_upgrade_todt_organization_to_stage_2 = {

		}
		swap_ideas = {
			remove_idea = GER_organization_todt
			add_idea = GER_organization_todt_improved
		}
	}

			remove_idea = GER_organization_todt_stage_1

}

# Stage 3 - the Wehrmacht integration choice converts the civilian factory bonus
# into either a synthetic refinery bonus or a military factory bonus.
GER_upgrade_todt_organization_to_refinery = {
	if = {
		limit = {
			has_idea = GER_organization_todt
		}
		swap_ideas = {
			remove_idea = GER_organization_todt
			add_idea = GER_organization_todt_integrated_refinery
		}
	}
	else_if = {
		limit = {
			has_idea = GER_organization_todt_stage_1
		}
		swap_ideas = {
			remove_idea = GER_organization_todt_stage_1
			add_idea = GER_organization_todt_integrated_refinery
		}
	}
	else_if = {
		limit = {
			has_idea = GER_organization_todt_improved
		}
		swap_ideas = {
			remove_idea = GER_organization_todt_improved
			add_idea = GER_organization_todt_integrated_refinery
		}
	}
	else_if = {
		limit = {
			has_idea = GER_organization_todt_integrated_military
		}
		swap_ideas = {
			remove_idea = GER_organization_todt_integrated_military
			add_idea = GER_organization_todt_integrated_refinery
		}
	}
}

# Keep the Organization Todt concern in sync with Todt's completed focuses.
GER_update_todt_organization_stage = {
	if = {
		limit = {
			has_completed_focus = GER_integrate_organization_todt_into_wehrmacht
		}
		if = {
			limit = {
				has_country_flag = GER_todt_integration_refinery
			}
			GER_upgrade_todt_organization_to_refinery = yes
		}
		else_if = {
			limit = {
				has_country_flag = GER_todt_integration_military
			}
			GER_upgrade_todt_organization_to_military = yes
		}
	}
	else_if = {
		limit = {
			has_completed_focus = GER_autonomous_organization_todt
		}
		GER_upgrade_todt_organization_to_stage_2 = yes
	}
	else_if = {
		limit = {
			has_completed_focus = GER_form_organization_todt
		}
		GER_upgrade_todt_organization_to_stage_1 = yes
	}
}
