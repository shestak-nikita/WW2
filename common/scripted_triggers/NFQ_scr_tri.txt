	NFQ_IsFocusSelected=no  # No focus currently active

}
# NFQ_IsFocusSelected: Checks if a focus is currently selected.
# Uses a game modifier to determine if a focus is active (hacky, but works ig).
NFQ_IsFocusSelected = {
	set_temp_variable={NFQ_polcost=modifier@female_random_country_leader_chance}  # Get modifier value
	check_variable={NFQ_polcost > 1.99}  # If >1.99, assume no focus selected
}
